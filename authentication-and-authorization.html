    8. Authentication and Authorization

    Two distinct questions arise for each request:

    • Who is the user? This is authentication: the user has identified themselves with a username and password, and each subsequent request must prove that it comes from them;
    • Does the user have permission to access this page? This is authorization: in the case study, only a user with the role ADMIN manages doctors and clients.

    Since the server is stateless (Chapter 6), proof of identity must be included with every request. A token JWT ([JSON Web Token]) is used: after login, the server generates a token containing the user’s identity and signs it with a secret key. The token is stored in a cookie; the browser sends it back with every request; the server verifies the signature. The user can read their token but cannot modify it: any modification invalidates the signature.

    [ASP.NET Core] clearly distinguishes between the two issues:

    • authentication is handled by middleware ([UseAuthentication]) and an authentication scheme: in this case, the [JwtBearer] scheme, provided by the [NuGet] [Microsoft.AspNetCore.Authentication.JwtBearer] package. For each request, it looks for a token, verifies it, and derives the user from it, [HttpContext.User];
    • Authorization is handled by the [[Authorize]] attribute, verified by the [UseAuthorization] middleware: it compares the action’s requirements with what is known about the user.

    The examples in this chapter use two [NuGet] packages: [Microsoft.AspNetCore.Authentication.JwtBearer] and [BCrypt.Net-Next] (password hashing). They are declared in the project file; the command [dotnet build] (or [dotnet run]) downloads them from nuget.org during the first compilation: an Internet connection is therefore required.

    namespace Exemple25.Auth;

public record Utilisateur(string Login, string Nom, string Role, string HachageMotDePasse);

// users, in memory (the next chapter will store them in a database)
public class UtilisateursService
{
    private readonly List<Utilisateur> utilisateurs =
    [
         // We do not store JAMAIS a password, only its hash
        Creer("marie", "Marie Pélissier", "USER", "marie"),
        Creer("admin", "Administrateur", "ADMIN", "admin"),
    ];

    public Utilisateur? Chercher(string login) =>
        utilisateurs.Find(u => u.Login == login);

     // bcrypt, with a cost of 10: 2^10 computation cycles, intentionally slow
    private static Utilisateur Creer(string login, string nom, string role, string motDePasse) =>
        new(login, nom, role, BCrypt.Net.BCrypt.HashPassword(motDePasse, 10));
}
    • line 3: a user, along with their role;
    • lines 10–12: a password is never stored; only its hash is retained—a fingerprint calculated by a one-way function. It is possible to verify that a password matches a fingerprint, but not to recover the password from the fingerprint;
    • lines 19–20: [BCrypt.Net.BCrypt.HashPassword(motDePasse, 10)] calculates the hash using the [bcrypt] algorithm. The 10 represents the “cost”: [bcrypt] is intentionally slow (2¹⁰ calculation cycles) to discourage anyone from trying millions of passwords. Two hashes of the same password are different (a random “salt” is added): [BCrypt.Verify] is responsible for verifying them.

    using System.Text;
using Microsoft.IdentityModel.JsonWebTokens;
using Microsoft.IdentityModel.Tokens;

namespace Exemple25.Auth;

public class AuthService(UtilisateursService utilisateurs, IConfiguration configuration)
{
     // the name of the cookie that carries the token
    public const string Cookie = "jeton";

     // the key that signs the tokens, read from appsettings.json (JWT section)
    public static SymmetricSecurityKey Cle(IConfiguration configuration) =>
        new(Encoding.UTF8.GetBytes(configuration["Jwt:Secret"]!));

     // null if the login is unknown or the password is incorrect
    public Utilisateur? Valider(string? login, string? motDePasse)
    {
        var utilisateur = utilisateurs.Chercher(login ?? "");
        if (utilisateur is null || motDePasse is null)
        {
            return null;
        }
        return BCrypt.Net.BCrypt.Verify(motDePasse, utilisateur.HachageMotDePasse) ? utilisateur : null;
    }

     // the user’s token JWT, signed with the secret key
    public string CreerJeton(Utilisateur utilisateur)
    {
        var descripteur = new SecurityTokenDescriptor
        {
             // the token’s content: the “claims” (statements about the user)
            Claims = new Dictionary<string, object>
            {
                ["sub"] = utilisateur.Login,
                ["nom"] = utilisateur.Nom,
                ["role"] = utilisateur.Role,
            },
            Expires = DateTime.UtcNow.AddMinutes(configuration.GetValue<int>("Jwt:DureeMinutes")),
            SigningCredentials = new SigningCredentials(Cle(configuration), SecurityAlgorithms.HmacSha256),
        };
        return new JsonWebTokenHandler().CreateToken(descripteur);
    }
}

    • line 10: the name of the cookie that carries the token;
    • Lines 13–14: The signature key, generated from the text of [appsettings.json]. [IConfiguration] provides access to the configuration; ["Jwt:Secret"] refers to the [Secret] property in the [Jwt] section. The same key is used to sign and verify tokens (symmetric algorithm);
    • lines 17–24: verification of the username and password;
    • lines 33–37: the token’s content, the claims (statements about the user). [sub] ([subject]) is the conventional name for the user’s identifier. We include what the pages need (name, role) so we don’t have to query the database with every request;
    • line 39: the token’s expiration date;
    • line 40: the signature key and algorithm;

    using System.Text.Encodings.Web;
using System.Text.Unicode;
using Exemple25.Auth;
using Microsoft.AspNetCore.Authentication.JwtBearer;
using Microsoft.Extensions.FileProviders;
using Microsoft.IdentityModel.Tokens;

var builder = WebApplication.CreateBuilder(args);
builder.Services.AddControllersWithViews();
builder.Services.AddSingleton(HtmlEncoder.Create(UnicodeRanges.All));
builder.Services.AddSingleton<UtilisateursService>();
builder.Services.AddSingleton<AuthService>();

// authentication: a JWT token, verified with each request
builder.Services
    .AddAuthentication(JwtBearerDefaults.AuthenticationScheme)
    .AddJwtBearer(options =>
    {
        options.MapInboundClaims = false; // Keep the claim names as-is (“sub,” “role,” etc.)
        options.TokenValidationParameters = new TokenValidationParameters
        {
            ValidateIssuerSigningKey = true,
            IssuerSigningKey = AuthService.Cle(builder.Configuration), // The signature must be ours
            ValidateLifetime = true,                                   // The token must not have expired
            ValidateIssuer = false,
            ValidateAudience = false,
            NameClaimType = "nom",  // User.Identity.Name
            RoleClaimType = "role", // [Authorize(Roles = ...)], User.IsInRole(...)
        };
        options.Events = new JwtBearerEvents
        {
             // 1. The token is read from the cookie (not from the Authorization header)
            OnMessageReceived = contexte =>
            {
                contexte.Token = contexte.Request.Cookies[AuthService.Cookie];
                return Task.CompletedTask;
            },
             // 2. Protected page without a valid token: redirect to the login page
            OnChallenge = contexte =>
            {
                contexte.HandleResponse(); // We generate the response ourselves (no 401)
                contexte.Response.Redirect("/login");
                return Task.CompletedTask;
            },
        };
    });
builder.Services.AddAuthorization();

var app = builder.Build();

app.UseStaticFiles(new StaticFileOptions
{
    FileProvider = new PhysicalFileProvider(Path.GetFullPath(Path.Combine(builder.Environment.ContentRootPath, "..", "commun", "bootstrap"))),
    RequestPath = "/bootstrap",
});
app.UseAuthentication(); // Who is this? -> HttpContext.User
app.UseAuthorization();  // Do they have permission? -> [Authorize]

app.MapControllers();
app.Run();

    Let’s comment on this code:

    • lines 16–17: We declare the [JwtBearer] authentication schema and set it as the default schema;
    • line 19: By default, [JwtBearer] renames certain claims using long names inherited from Microsoft technologies ([role] would become [http://schemas.microsoft.com/ws/2008/06/identity/claims/role]). We retain the token names;
    • Lines 20–26: The checks performed on each token: the signature (using our key) and the expiration date. Our tokens do not specify either the issuer ([issuer]) or the recipient ([audience]), so we do not verify them;
    • lines 27–28: which claim contains the user’s name ([User.Identity.Name]), and which one contains their role ([[Authorize(Roles = ...)]], [User.IsInRole(...)]);
    • lines 33–36: the events from [JwtBearer] allow its behavior to be customized. By default, it looks for the token in the [Authorization: Bearer ...] header of the request: this is how a [JavaScript] client or a mobile app operates. A browser does not add this header on its own: the token is retrieved from the cookie;
    • lines 39–42: the challenge occurs when a protected page is requested without a valid token. By default, the response is a 401 error with no content: useful for a [JavaScript] client, but confusing for a user. The user is redirected to the login page. [HandleResponse()] tells [JwtBearer] that the response is being handled;
    • lines 56–57: the two middleware components, in this order: first determine who the user is, then verify their permissions.

    using Exemple25.Auth;
using Microsoft.AspNetCore.Mvc;

namespace Exemple25.Controllers;

public class AuthController(AuthService auth, IConfiguration configuration) : Controller
{
    [HttpGet("/")]
    public IActionResult Accueil() => Redirect("/profil");

    [HttpGet("login")]
    public IActionResult Formulaire() => View("Login");

    [HttpPost("login")]
    public IActionResult Connexion(string? login, string? motDePasse)
    {
        var utilisateur = auth.Valider(login, motDePasse);
        if (utilisateur is null)
        {
             // A deliberately vague message: it doesn’t specify whether it’s the username or the password
            ViewData["Erreur"] = "Login ou mot de passe incorrect.";
            ViewData["Login"] = login;
            return View("Login");
        }
         // the token, stored in a cookie that the page’s JavaScript cannot read
        Response.Cookies.Append(AuthService.Cookie, auth.CreerJeton(utilisateur), new CookieOptions
        {
            HttpOnly = true,
            SameSite = SameSiteMode.Lax,
            MaxAge = TimeSpan.FromMinutes(configuration.GetValue<int>("Jwt:DureeMinutes")),
        });
        return Redirect("/profil");
    }

     // Logging out changes the user’s status: it’s a POST
    [HttpPost("logout")]
    public IActionResult Deconnexion()
    {
        Response.Cookies.Delete(AuthService.Cookie);
        return Redirect("/login");
    }
}

    Let’s comment on this code:

    • lines 14–23: the login. If the login fails, the form is redisplayed with a deliberately vague message: saying “this login does not exist” would help an attacker find valid logins;
    • lines 26–32: if successful, the token is stored in a cookie named [HttpOnly]; the page’s [JavaScript] cannot read it, which protects the token if a malicious script were to execute on the page. [SameSite = Lax]: The cookie is not included in POST requests sent from another site, which protects against request forgery ([CSRF]). The cookie’s lifetime matches that of the token;
    • Lines 36–40: Logging out deletes the cookie ([Cookies.Delete]). This is a POST: it changes the user’s status (see Example 21).

    8.3. Example 27: CAPTCHA and attempt limits

    A login page attracts bots: programs that try thousands of login/password combinations. Two complementary safeguards:

    • a CAPTCHA: an image of distorted characters that a human can easily read but that a simple bot cannot decipher;
    • an attempt limit: after a few failed attempts, the address IP is blocked for a while.

namespace Exemple27.Auth;

// a CAPTCHA: its ID and its image (SVG)
public record Captcha(string Id, string Svg);

// singleton: pending captchas are stored in memory by the server
public class CaptchaService
{
    private static readonly TimeSpan DureeVie = TimeSpan.FromMinutes(10);

     // ID -> (expected response, expiration date).
     // ConcurrentDictionary: a dictionary that can be used by multiple simultaneous requests
    private readonly ConcurrentDictionary<string, (string Texte, DateTime Expire)> enAttente = new();

    public Captcha Generer()
    {
        var texte = DessinCaptcha.TexteAleatoire(5);
        var id = Guid.NewGuid().ToString(); // a random identifier, impossible to guess
        enAttente[id] = (texte, DateTime.UtcNow + DureeVie);
        return new Captcha(id, DessinCaptcha.Svg(texte));
    }

     // the CAPTCHA is CONSOMMÉ in all cases: an image is used only once
    public bool Verifier(string? id, string? saisie)
    {
        if (id is null || saisie is null || !enAttente.TryRemove(id, out var attendu))
        {
            return false;
        }
        return attendu.Expire > DateTime.UtcNow
            && string.Equals(attendu.Texte, saisie.Trim(), StringComparison.OrdinalIgnoreCase);
    }
}

    Let’s break down this code:

    • line 15: the expected responses are stored by the server under a random identifier. The page contains only the image and the identifier: the response never leaves the server. The service is a singleton, used by concurrent requests: [ConcurrentDictionary] is a dictionary designed for this purpose;
    • lines 17–22: a five-character text string is randomly selected, stored, and drawn as an image. The drawing is handled by the class [DessinCaptcha] (file [Auth/DessinCaptcha.cs], taken from the case study and commented on line by line): each character is a set of lines that is enlarged, tilted, and randomly distorted, and the image is blurred by three curves. Characters that are easily confused (0 and O, 1, I and L) are excluded;
    • lines 26–33: the verification process consumes the CAPTCHA ([TryRemove] removes it from the dictionary): an image is used only once; a bot cannot reuse a solution it has already found. Uppercase letters do not count.

    namespace Exemple27.Auth;

// singleton: counts failed login attempts for each address IP
public class LimiteurService
{
    private const int MaxEchecs = 3;
    private static readonly TimeSpan DureeBlocage = TimeSpan.FromMinutes(1);

    private class Compteur
    {
        public int Echecs;
        public DateTime BloqueJusqua;
    }

    private readonly Dictionary<string, Compteur> compteurs = [];
    private readonly Lock verrou = new(); // multiple simultaneous requests: only one access at a time

     // the number of seconds remaining in the lockout period (0: not locked out)
    public int SecondesBlocage(string ip)
    {
        lock (verrou)
        {
            if (!compteurs.TryGetValue(ip, out var c)) return 0;
            var reste = c.BloqueJusqua - DateTime.UtcNow;
            return reste > TimeSpan.Zero ? (int)Math.Ceiling(reste.TotalSeconds) : 0;
        }
    }

    public void Echec(string ip)
    {
        lock (verrou)
        {
            if (!compteurs.TryGetValue(ip, out var c))
            {
                c = new Compteur();
                compteurs[ip] = c;
            }
            c.Echecs++;
            if (c.Echecs >= MaxEchecs)
            {
                c.BloqueJusqua = DateTime.UtcNow + DureeBlocage;
                c.Echecs = 0; // After the lock, the count resets to zero
            }
        }
    }

    public void Reussite(string ip)
    {
        lock (verrou)
        {
            compteurs.Remove(ip);
        }
    }
}
    • lines 6–7: three failed attempts block the address IP for one minute;
    • line 16: the counter dictionary is shared by all queries. The instruction [lock (verrou) { ... }] ensures that only one query at a time executes the block: two simultaneous failures cannot be lost (see [Interlocked], Chapter 6). [Lock] is the lock type for [.NET] 9 and subsequent lines;
    • lines 19–25: the remaining lock time;
    • lines 29–42: a failure is counted; on the third failure, the address is blocked;
    • lines 47–51: a success resets the counter to zero.

    using Exemple27.Auth;
using Microsoft.AspNetCore.Mvc;

namespace Exemple27.Controllers;

public class AuthController(AuthService auth, CaptchaService captchas, LimiteurService limiteur,
    IConfiguration configuration) : Controller
{
    [HttpGet("/")]
    public IActionResult Accueil() => Redirect("/profil");

    [HttpGet("login")]
    public IActionResult Formulaire() => Login(null, null);

    [HttpPost("login")]
    public IActionResult Connexion(string? login, string? motDePasse, string? captchaId, string? captcha)
    {
         // the client's address IP
        var ip = HttpContext.Connection.RemoteIpAddress?.ToString() ?? "?";

         // 1. A blocked address is not even listened to
        var secondes = limiteur.SecondesBlocage(ip);
        if (secondes > 0)
        {
            return Login(login, $"Trop de tentatives : réessayez dans {secondes} s.");
        }
         // 2. First the CAPTCHA (low-cost), then the password (bcrypt, high-cost)
        if (!captchas.Verifier(captchaId, captcha))
        {
            limiteur.Echec(ip);
            return Login(login, "Le code de l'image est incorrect.");
        }
        var utilisateur = auth.Valider(login, motDePasse);
        if (utilisateur is null)
        {
            limiteur.Echec(ip);
            return Login(login, "Login ou mot de passe incorrect.");
        }
        limiteur.Reussite(ip);
        Response.Cookies.Append(AuthService.Cookie, auth.CreerJeton(utilisateur), new CookieOptions
        {
            HttpOnly = true,
            SameSite = SameSiteMode.Lax,
            MaxAge = TimeSpan.FromMinutes(configuration.GetValue<int>("Jwt:DureeMinutes")),
        });
        return Redirect("/profil");
    }

    [HttpPost("logout")]
    public IActionResult Deconnexion()
    {
        Response.Cookies.Delete(AuthService.Cookie);
        return Redirect("/login");
    }

     // Each time the form is displayed, a NOUVEAU CAPTCHA is generated
    private ViewResult Login(string? login, string? erreur)
    {
        ViewData["Login"] = login;
        ViewData["Erreur"] = erreur;
        return View("Login", captchas.Generer());
    }
}
